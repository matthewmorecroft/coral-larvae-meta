    pop_genetic == "1" & pop_genomic == "1" & experimental == "1" ~ "genom_experimental_comb",
   pop_genetic == "1" & biophys == "1" ~ "gene_model_comb",
   experimental == "1" & biophys == "1" ~ "experimental_model_comb",
   pop_genetic == "1" & experimental == "1" ~ "experimental_genetic_comb",

   pop_genetic == "1" ~ "Genetics",
   biophys == "1" ~ "Biophysical modelling",

just_specificity_sum <- biophys_singleparam %>% 
  summarise(prop_coralspec = ( sum(just_coral_specific, na.rm = TRUE) /137 ) * 100,
            prop_reprospec = ( sum(just_repro_mode_specific, na.rm = TRUE) /137 ) * 100,
            prop_orderspec = ( sum(just_order_specific, na.rm = TRUE) /137 ) * 100,
            prop_familyspec = ( sum(just_family_specific, na.rm = TRUE) /137 ) * 100,
            prop_genusspec = ( sum(just_genus_specific, na.rm = TRUE) /137 ) * 100,
            prop_speciesspec = ( sum(just_species_specific, na.rm = TRUE) /137 ) * 100,
            G_prop_gencoral = ( sum(is.na(just_coral_specific)) /137 ) * 100,
            G_prop_genrepro = ( sum(is.na(just_repro_mode_specific)) /137 ) * 100,
            G_prop_genorder = ( sum(is.na(just_order_specific)) /137 ) * 100,
            G_prop_genfamily = ( sum(is.na(just_family_specific)) /137 ) * 100,
            G_prop_gengenus = ( sum(is.na(just_genus_specific)) /137 ) * 100,
            G_prop_genspecies = ( sum(is.na(just_species_specific)) /137 ) * 100,
            speccoral_dataprop = ( sum(just_coral_specific, na.rm = TRUE) / (137 - sum(is.na(just_coral_specific)) ) ) * 100,
            specrepro_dataprop = ( sum(just_repro_mode_specific, na.rm = TRUE) / (137 - sum(is.na(just_repro_mode_specific)) ) ) * 100,
            specorder_dataprop = ( sum(just_order_specific, na.rm = TRUE) / (137 - sum(is.na(just_order_specific)) ) ) * 100,
            specfamily_dataprop = ( sum(just_family_specific, na.rm = TRUE) / (137 - sum(is.na(just_family_specific)) ) ) * 100,
            specgenus_dataprop = ( sum(just_genus_specific, na.rm = TRUE) / (137 - sum(is.na(just_genus_specific)) ) ) * 100,
            specspecies_dataprop = ( sum(just_species_specific, na.rm = TRUE) / (137 - sum(is.na(just_species_specific)) ) ) * 100,
 ) %>% 
  pivot_longer(cols = everything(),
               names_to = "level",
               values_to = "proportion") %>% 
   mutate(generic_code = case_when(
    grepl("G_", level) ~ "2",
    grepl("_dataprop", level) ~ "3",
    .default = "1"
  )) %>% 
  mutate(generic = case_when(
    generic_code == 2 ~ "proportion of generic corals for this level",
    generic_code == 3 ~ "from total excluding generic corals for this level",
    .default = "from total including generic corals"
  )) %>% 
  mutate(sumlevel = case_when(
    grepl("coral", level) ~ "coral",
    grepl("repro", level) ~ "reproductive-mode",
    grepl("order", level) ~ "order",
    grepl("family", level) ~ "family",
    grepl("genus", level) ~ "genus",
    grepl("species", level) ~ "species"
    ))
